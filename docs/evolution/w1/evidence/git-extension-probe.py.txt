import pathlib,tempfile,subprocess,json
root=pathlib.Path(tempfile.mkdtemp(prefix='hseos-git-extension-probe-',dir='/tmp'))
def git(*args):return subprocess.run(['git','-C',str(root),*args],capture_output=True,text=True,check=True)
git('init','-b','task/git-probe');git('config','user.name','Fixture');git('config','user.email','fixture@example.invalid')
(root/'index.js').write_text('before\n');(root/'.gitattributes').write_text('*.js filter=canary\n');git('add','.');git('commit','-m','test: baseline')
marker=root/'marker';script=root/'filter.sh';script.write_text('#!/bin/sh\nprintf effect >> "'+str(marker)+'"\ncat\n');script.chmod(0o700);git('config','filter.canary.clean',str(script))
(root/'index.js').write_text('after\n')
result=subprocess.run(['git','--no-optional-locks','-C',str(root),'diff','--quiet','HEAD','--','index.js'],capture_output=True)
print(json.dumps({'directory':str(root),'diff_exit':result.returncode,'filter_executed':marker.exists()}))
