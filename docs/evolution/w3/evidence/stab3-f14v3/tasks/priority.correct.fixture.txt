const BY_SEVERITY = { critical: 'P1', high: 'P2', medium: 'P3', low: 'P4' };
function classifyPriority(ticket) {
  if ((ticket.tags || []).includes('billing')) return 'P2';
  const priority = BY_SEVERITY[ticket.severity];
  if (!priority) throw new Error(`unknown severity: ${ticket.severity}`);
  return priority;
}
module.exports = classifyPriority;
